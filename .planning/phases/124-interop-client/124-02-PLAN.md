```typescript
// bbj-vscode/src/language/java-interop-program-types.ts
export interface ProgramPosition { line: number; character: number }      // 0-based, UTF-16 code units
export interface ProgramRange { start: ProgramPosition; end: ProgramPosition }
export interface ProgramTextEdit { range: ProgramRange; newText: string }
export type ProgramSeverity = 'ERROR' | 'WARNING' | 'INFO';
export interface ProgramDiagnostic {
    line: number;                 // 1-based line in the result's own text; 0 = no location (kept, never coerced)
    originalLineNumber: string;   // '' when unknown
    severity: ProgramSeverity;
    message: string;              // plain text, sanitised
}
export type FormatSettingValue = string | number | boolean;
export interface FormatProgramParams {
    text: string;                 // the live editor text, passed by the caller; the client never reads files
    version: string;              // always sent; the echo is verified
    canonicalName?: string;
    settings?: Record<string, FormatSettingValue>;
    allowDenum?: boolean;
    range?: ProgramRange;
}
export interface DocumentFormatResult { scope: 'document'; text: string; diagnostics: ProgramDiagnostic[]; denumbered: boolean; version: string }
export interface RangeFormatResult { scope: 'range'; edits: ProgramTextEdit[]; diagnostics: ProgramDiagnostic[]; denumbered: boolean; version: string }
export type FormatProgramResult = DocumentFormatResult | RangeFormatResult;
export interface DenumProgramParams { text: string; version: string; canonicalName?: string }
export interface DenumProgramResult { text: string; diagnostics: ProgramDiagnostic[]; denumbered: boolean; version: string }
export type ProgramMethod = 'formatProgram' | 'denumProgram';
export interface ProgramSettingProblem { setting: string; message: string }
export type ProgramFailureKind =
    | 'parser-exception' | 'size-cap' | 'service-unavailable' | 'protected-program'
    | 'denum-needed' | 'format-failed' | 'invalid-params' | 'transport';
export type ProgramOutcome<R> =
    | { kind: 'ok'; result: R }
    | { kind: 'cancelled' }
    | { kind: 'timeout'; origin: 'client' | 'peer' }
    | { kind: 'unavailable'; reason: 'method-not-found' | 'not-reachable' }
    | { kind: 'invalid-settings'; problems: ProgramSettingProblem[] }
    | { kind: 'mixed-numbering'; line: number | undefined }
    | { kind: 'failed'; failure: ProgramFailureKind; code: number | undefined; message: string }
    | { kind: 'malformed-result'; reason: string };
export const formatProgramRequest: RequestType<FormatProgramParams, unknown, null>; // method 'formatProgram'
export const denumProgramRequest: RequestType<DenumProgramParams, unknown, null>;   // method 'denumProgram'

// bbj-vscode/src/language/java-program-guard.ts
export const PROGRAM_TEXT_RELATIVE_FACTOR = 4;
export const PROGRAM_TEXT_SLACK = 65536;                 // 64 KiB
export const PROGRAM_TEXT_ABSOLUTE_CAP = 16777216;       // 16 MiB
export const MAX_PROGRAM_DIAGNOSTICS = 500;
export const MAX_PROGRAM_DIAGNOSTIC_MESSAGE_LENGTH = 1024;
export const MAX_ORIGINAL_LINE_NUMBER_LENGTH = 32;
export type ProgramGuardResult<R> = { ok: true; value: R } | { ok: false; reason: string };
export function allowedProgramTextLength(requestText: string): number;
export function sanitizePeerText(text: string, limit: number): string;
export function programLineLengths(text: string): number[];
export function sanitizeProgramDiagnostics(raw: unknown, lineCount: number): ProgramGuardResult<ProgramDiagnostic[]>;
export function validateFormatResult(request: FormatProgramParams, raw: unknown): ProgramGuardResult<FormatProgramResult>;
export function validateDenumResult(request: DenumProgramParams, raw: unknown): ProgramGuardResult<DenumProgramResult>;
```

- range: keys `edits, diagnostics, denumbered, version`; `text` absent; already-formatted range -> `edits: []`
- text `'if a then print 1\n  x=1\nrem y\n'`, request range (0,0)-(2,0) -> one edit (0,0)-(2,0), newText ends `\n`

- DENUM `0010 print 1\n0020 goto 0010\n` -> `denumbered: true`, text `L10: print 1\ngoto L10\n`
- DENUM with a syntax error -> `diagnostics: [{line:1, originalLineNumber:"0010", severity:"ERROR", message:"syntax error"}, ...]`
